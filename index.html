Below is the source code for the console-based game developed as part of the CC102 coursework.


// 173

import java.util.Scanner;

public class Array {
    public static void main(String[] args) {
        Scanner input = new Scanner(System.in);
        
        int size = 15;
        int[] originalArray = new int[size];
        int[] rotatedArray = new int[size];

        
        System.out.println("======================ARRAY LEFT ROTATION APPLICATION======================");
        System.out.println("Please enter " + size + " integers ranging from 0 to 100:\n");

        for (int i = 0; i  < size; i++) {
            while (true) {
                System.out.print("Enter value for index " + i + ": ");
                if (input.hasNextInt()) {
                    int value = input.nextInt();
                    if (value >= 0 && value <= 100) {
                        originalArray[i] = value;
                        break;
                    } else {
                        System.out.println("  -> Error: Value must be between 0 and 100. Try again.");
                    }
                } else {
                    System.out.println("  -> Error: Invalid input. Please enter a valid integer.");
                    input.next();
                }
            }
        }

        for (int i = 0; i < size; i++) {
            rotatedArray[i] = originalArray[i];
        }

        int largestValue = originalArray[0];
        for (int i = 1; i < size; i++) {
            if (originalArray[i] > largestValue) {
                largestValue = originalArray[i];
            }
        }

       
        int firstElement = rotatedArray[0];
        
        for (int i = 0; i < size - 1; i++) {
            rotatedArray[i] = rotatedArray[i + 1];
        }
        
        rotatedArray[size - 1] = firstElement;

        // Step 5: Display Results
        System.out.println("==================RESULTS SUMMARY==================");
        
      
        System.out.print("Original Array: [");
        for (int i = 0; i < size; i++) {
            System.out.print(originalArray[i] + (i < size - 1 ? ", " : ""));
        }
        System.out.println("]");


        System.out.print("Rotated Array:  [");
        for (int i = 0; i < size; i++) {
            System.out.print(rotatedArray[i] + (i < size - 1 ? ", " : ""));
        }
        System.out.println("]");

        System.out.println("Largest Value:  " + largestValue);
        System.out.println("==========================================");

        input.close();
    }
}
