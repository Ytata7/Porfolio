Below is the console-based game source code developed as the final requirement for CC102 (Introduction to Computer Programming).



import java.util.Random;
import java.util.Scanner;

public class SimpleGame {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int stone = random.nextInt(3) + 1;
        int guess;

        System.out.println("================================");
        System.out.println("     3 CUP GUESSING GAME");
        System.out.println("================================");

        System.out.println("There is a stone hidden under one cup.");
        System.out.println("Guess where the stone is!");

        System.out.println();
        System.out.println("   [1]       [2]       [3]");
        System.out.println("   CUP       CUP       CUP");

        System.out.print("\nChoose a cup (1-3): ");
        guess = input.nextInt();

        if (guess < 1 || guess > 3) {
            System.out.println("Invalid choice! Please choose 1, 2, or 3.");
        } 
        else if (guess == stone) {
            System.out.println("\nCongratulations! You found the stone!");
            System.out.println("You WIN!");
        } 
        else {
            System.out.println("\nSorry! Wrong cup.");
            System.out.println("The stone was under cup number " + stone);
            System.out.println("You LOSE!");
        }

        System.out.println("\nThank you for playing!");
        
        input.close();
    }
}
